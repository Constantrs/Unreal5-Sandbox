#pragma once

#include "CoreMinimal.h"
#include "GameFramework/GameMode.h"
#include "TimerManager.h"
#include "SideScrollerAssetLoadSubsystem.h"
#include "SideScrollerGameMode.generated.h"

UENUM(BlueprintType)
enum class ESideScrollerEnemyLoadState : uint8
{
    Idle, LoadingStageConfig, LoadingDefinitions, LoadingMeshes, Spawning, Ready, Failed, Cancelled
};

/** Orchestrates UI and enemy spawning; resource ownership belongs to AssetLoadSubsystem. */
UCLASS(Blueprintable)
class SIDESCROLLERRUNTIME_API ASideScrollerGameMode : public AGameMode
{
    GENERATED_BODY()

public:
    ASideScrollerGameMode();
    virtual void InitGame(const FString& MapName, const FString& Options, FString& ErrorMessage) override;

    UFUNCTION(BlueprintPure, Category = "Stage|Enemies")
    ESideScrollerEnemyLoadState GetEnemyLoadState() const;

    UFUNCTION(BlueprintPure, Category = "Stage|Enemies")
    FString GetEnemyLoadError() const { return EnemyLoadError; }

    UFUNCTION(BlueprintPure, Category = "Stage|Assets")
    int32 GetAssetLoadRequestId() const { return AssetLoadRequestId; }

    /** Does not destroy enemies. Unfinished loads are cancelled; cannot release mid-spawn. */
    UFUNCTION(BlueprintCallable, Category = "Stage|Assets")
    bool ReleaseStageAssets();

protected:
    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Stage")
    TObjectPtr<USideScrollerStageConfigAsset> StageConfigs;

    // Inspection only; these do not keep catalog/table resources alive.
    UPROPERTY(Transient, VisibleInstanceOnly, BlueprintReadOnly, Category = "Stage|Enemies")
    TWeakObjectPtr<USideScrollerEnemyDataAsset> EnemyAssets;

    UPROPERTY(Transient, VisibleInstanceOnly, BlueprintReadOnly, Category = "Stage|Enemies")
    TWeakObjectPtr<UDataTable> EnemySpawnTable;

    UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Stage|Debug")
    bool bUseDebugLoading = false;

    UPROPERTY(VisibleInstanceOnly, BlueprintReadOnly, Category = "Stage")
    FName StageId;

    UFUNCTION(BlueprintImplementableEvent, Category = "Stage|Enemies")
    void OnEnemiesReady();

    UFUNCTION(BlueprintImplementableEvent, Category = "Stage|Enemies")
    void OnEnemyLoadFailed(const FString& Reason);

    void StartStageConfigLoading();
    void StartEnemyLoading(USideScrollerEnemyDataAsset* Catalog, UDataTable* Table);
    virtual void BeginPlay() override;
    virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;

private:
    USideScrollerAssetLoadSubsystem* BindAssetLoader();
    void TryStartLoading();
    void SpawnEnemies();
    void UpdateLoadingProgress();
    void FailEnemyLoading(const FString& Reason);
    void ClearAssetRequest();

    UFUNCTION() void HandleAssetsReady(int32 RequestId);
    UFUNCTION() void HandleAssetsFailed(int32 RequestId, const FString& Reason);
    UFUNCTION() void HandleAssetsCancelled(int32 RequestId);

    FTimerHandle WaitForPlayerTimer;
    FTimerHandle LoadProgressTimer;
    int32 AssetLoadRequestId = 0;
    bool bEndingPlay = false;
    ESideScrollerEnemyLoadState EnemyLoadState = ESideScrollerEnemyLoadState::Idle;
    FString EnemyLoadError;
    FString StageOptionError;

    // Only retained while synchronous spawn/Blueprint initialization is executing.
    UPROPERTY(Transient) FSideScrollerStageLoadResult SpawnData;
    UPROPERTY(Transient) TArray<TWeakObjectPtr<AActor>> SpawnedEnemies;
};
